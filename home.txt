  <script>
    function toggleSlide(type) {
      closeSlides();
      document.getElementById('overlay').classList.add('show');
      if (type === 'location') document.getElementById('locationSlide').classList.add('slide-open');
      else if (type === 'notification') document.getElementById('notificationSlide').classList.add('slide-open');
      else if (type === 'user') document.getElementById('userSlide').classList.add('slide-open');
      else if (type === 'menu') document.getElementById('menuSlide').classList.add('slide-open');
    }

    function closeSlides() {
      document.getElementById('overlay').classList.remove('show');
      ['menuSlide','locationSlide','notificationSlide','userSlide'].forEach(id=>{
        document.getElementById(id).classList.remove('slide-open');
      });
    }

    document.addEventListener('keydown', (e) => { if(e.key==="Escape") closeSlides(); });

    const adSlider = document.getElementById("adSlider");
    adSlider.addEventListener("wheel", function(e) { e.preventDefault(); adSlider.scrollLeft += e.deltaY*3; });

    const slides = document.querySelectorAll(".slide");
    const dotsContainer = document.getElementById("sliderDots");
    slides.forEach((_,index)=>{ const dot=document.createElement("div"); dot.classList.add("dot"); dot.addEventListener("click",()=>{ adSlider.scrollLeft = index*adSlider.clientWidth; }); dotsContainer.appendChild(dot); });
    function updateDots(){ const index = Math.round(adSlider.scrollLeft/adSlider.clientWidth); document.querySelectorAll(".dot").forEach((dot,i)=>dot.classList.toggle("active",i===index)); }
    adSlider.addEventListener("scroll", updateDots); updateDots();

    (function setupMostSearchedTabs(){
      const tabs = document.querySelectorAll('.most-searched .car-tabs .tab');
      const grids = document.querySelectorAll('.most-searched .cars-grid');
      function show(tabName){
        tabs.forEach(t=>t.classList.toggle('active',t.dataset.tab===tabName));
        grids.forEach(g=>{ g.style.display = g.id===tabName?'grid':'none'; });
      }
      tabs.forEach(t=>t.addEventListener('click',()=>show(t.dataset.tab)));
      show('suv');
    })();
